cmake_minimum_required(VERSION 3.16)
project(FinalBuildInventory VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(FBS_BUILD_TESTS "Build module tests" ON)
option(FBS_BUILD_EXAMPLES "Build the example" ON)
add_library(fbs_inventory src/inventory.c)
add_library(fbs::inventory ALIAS fbs_inventory)
set_target_properties(fbs_inventory PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES
  C_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON EXPORT_NAME inventory)
target_include_directories(fbs_inventory PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NOT MSVC)
  target_link_libraries(fbs_inventory PUBLIC m)
  target_compile_options(fbs_inventory PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(FBS_BUILD_TESTS AND BUILD_TESTING)
  add_executable(fbs_test_inventory tests/test_inventory.c)
  set_target_properties(fbs_test_inventory PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_test_inventory PRIVATE fbs::inventory)
  if(NOT MSVC)
    target_compile_options(fbs_test_inventory PRIVATE -UNDEBUG -ffp-contract=off)
  endif()
  add_test(NAME inventory COMMAND fbs_test_inventory)
  set_tests_properties(inventory PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/tests)
endif()
if(FBS_BUILD_EXAMPLES)
  add_executable(fbs_inventory_example examples/basic.c)
  set_target_properties(fbs_inventory_example PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_inventory_example PRIVATE fbs::inventory)
  if(BUILD_TESTING)
    add_test(NAME inventory_example COMMAND fbs_inventory_example)
  endif()
endif()
install(TARGETS fbs_inventory EXPORT FinalBuildInventoryTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT FinalBuildInventoryTargets NAMESPACE fbs::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FinalBuildInventory)
install(FILES LICENSE THIRD_PARTY_NOTICES DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-inventory)
install(DIRECTORY third_party/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-inventory/upstream OPTIONAL)
